function sourceRowKey(r){
  let game=String(v(r,'pff_GAMEID')||'').trim();
  let play=String(v(r,'pff_PLAYID')||'').trim();
  let sort=String(v(r,'pff_SORTORDER')||'').trim();
  let gsis=String(v(r,'pff_GSISPLAYID')||'').trim();
  if(game&&play)return `gp:${game}|${play}`;
  if(gsis)return `gsis:${gsis}`;
  return `fallback:${game}|${sort}|${v(r,'pff_OFFTEAM')}|${v(r,'pff_DEFTEAM')}|${v(r,'pff_SPECIALTEAMSTYPE')}`;
}
function mergePffSources(groups){
  const map=new Map();
  groups.flat().forEach(r=>{
    const key=sourceRowKey(r);
    if(!map.has(key)){map.set(key,{...r});return}
    const current=map.get(key);
    Object.keys(r).forEach(k=>{
      if((current[k]==null||String(current[k]).trim()==='')&&r[k]!=null&&String(r[k]).trim()!=='')current[k]=r[k];
    });
  });
  return [...map.values()];
}
function classifyCounts(rs){
  let team=prepConfig().rawTeam||teamSettings.teamCode||'';
  let c={};
  const add=k=>c[k]=(c[k]||0)+1;
  rs.forEach(r=>{
    let t=String(v(r,'pff_SPECIALTEAMSTYPE')||'').toUpperCase();
    if(!t)return;
    if(!team){add(t);return}
    let off=String(v(r,'pff_OFFTEAM')||'').toUpperCase()===String(team).toUpperCase();
    let def=String(v(r,'pff_DEFTEAM')||'').toUpperCase()===String(team).toUpperCase();
    if(t==='KICKOFF') add(off?'Kickoff Return':def?'Kickoff':'KICKOFF');
    else if(t==='PUNT') add(off?'Punt Return':def?'Punt':'PUNT');
    else if(t==='FIELD GOAL') add(off?'Field Goal':def?'FG Block':'FIELD GOAL');
    else if(t==='EXTRA POINT') add(off?'PAT':def?'PAT Block':'EXTRA POINT');
    else add(t);
  });
  return c;
}

async function loadPffFromSupabase(options={}){
  setDataSourceStatus(`Loading ${prepConfig().label} PFF special-teams files from Supabase…`);
  try{
    const results=await Promise.all(activePffFiles().map(async candidates=>{
      let lastStatus='';
      for(const name of candidates){
        const response=await fetch(pffStorageUrl(name),{cache:'no-store'});
        if(response.ok){
          const body=await response.text();
          const parsed=parseCSV(body);
          if(!parsed.length)throw new Error(`${name} contained no play rows`);
          return {name,rows:parsed};
        }
        lastStatus=`${name} returned ${response.status}`;
      }
      throw new Error(lastStatus||'PFF file not found');
    }));
    const rawCount=results.reduce((n,x)=>n+x.rows.length,0);
    rows=mergePffSources(results.map(x=>x.rows));
    const counts=classifyCounts(rows);
    const countText=Object.entries(counts).map(([k,n])=>`${k}: ${n}`).join(' • ');
    setDataSourceStatus(`${prepConfig().label} • Supabase live • 2 files • ${rawCount} raw rows • ${rows.length} unique plays${countText?' • '+countText:''}`);
    setup();
    playIndex=0;chartIndex=0;chartPositions={};
    if(options.render!==false)render();
    return true;
  }catch(err){
    console.error('Supabase PFF load failed:',err);
    rows=[];
    setup();
    setDataSourceStatus('PFF data could not load from Supabase: '+err.message,true);
    if(options.notice)alert('Could not load PFF data from Supabase: '+err.message);
    if(options.render!==false)render();
    return false;
  }
}

    };
  }

      allowed:true,

      current:currentByNum,

    };
  }

  const matches=historicalPlayersByNumber(num,season);

  // player owns the historical PFF stats. Then locate that SAME NAME anywhere
  // on the current roster instead of assuming the jersey number stayed fixed.

    const currentByName=currentRosterByName(historical.name);

      const changedNumber=!!(oldNum&&nowNum&&oldNum!==nowNum);
      return {
        status:changedNumber?'changed':'verified',

        allowed:true,

        current:currentByName,
        historical,
        changedNumber
      };
    }

    // Still show the historical player. This is useful for Week 1 prep and is
    // safer than attaching his stats to a different current player wearing the
    // same number.
    return {

    };

function prepDepthPlayerNamesForCategory(cat){
  const st=prepDepthChart?.specialTeams||{};
  const keys=cat==='kickReturners'?['KR']:
    cat==='puntReturners'?['PR']:
    cat==='kickers'?['PK','KO']:
    cat==='punters'?['PT']:
    cat==='koCoverageTacklers'?['KO']:['PT'];
  const seen=new Set(),out=[];
  keys.forEach(k=>(st[k]||[]).forEach(([num,name])=>{
    const key=nameKey(name);

  }));
  return out;
}

      body:JSON.stringify({action:'save',data:payload})

    const j=await r.json();

        repAssignments:d.repAssignments&&typeof d.repAssignments==='object'?d.repAssignments:{}
      }};
    }
    loadLookLibraryForOpponent(prepOpponent);
    CLOUD.lastLoadedAt=d.updatedAt||CLOUD.lastLoadedAt||'';

    saveRoster(false);

  }

function mergePffSources(groups){
  const map=new Map();
  groups.flat().forEach(r=>{
    const key=sourceRowKey(r);
    if(!map.has(key)){map.set(key,{...r});return}
    const current=map.get(key);
    Object.keys(r).forEach(k=>{
      if((current[k]==null||String(current[k]).trim()==='')&&r[k]!=null&&String(r[k]).trim()!=='')current[k]=r[k];
    });
  });
  return [...map.values()];
}
function classifyCounts(rs){
  let team=prepConfig().rawTeam||teamSettings.teamCode||'';
  let c={};
  const add=k=>c[k]=(c[k]||0)+1;
  rs.forEach(r=>{
    let t=String(v(r,'pff_SPECIALTEAMSTYPE')||'').toUpperCase();
    if(!t)return;
    if(!team){add(t);return}
    let off=String(v(r,'pff_OFFTEAM')||'').toUpperCase()===String(team).toUpperCase();
    let def=String(v(r,'pff_DEFTEAM')||'').toUpperCase()===String(team).toUpperCase();
    if(t==='KICKOFF') add(off?'Kickoff Return':def?'Kickoff':'KICKOFF');
    else if(t==='PUNT') add(off?'Punt Return':def?'Punt':'PUNT');
    else if(t==='FIELD GOAL') add(off?'Field Goal':def?'FG Block':'FIELD GOAL');
    else if(t==='EXTRA POINT') add(off?'PAT':def?'PAT Block':'EXTRA POINT');
    else add(t);
